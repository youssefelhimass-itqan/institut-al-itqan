-- A1. Fonctions SECURITY DEFINER présentes
SELECT routine_name, security_type
FROM information_schema.routines
WHERE routine_schema = 'public'
  AND routine_name IN ('get_my_role', 'get_my_acces_cours')
ORDER BY routine_name;
-- Attendu : 2 lignes, security_type = DEFINER

-- A2. RLS activée sur les tables concernées
SELECT relname, relrowsecurity
FROM pg_class
WHERE relnamespace = 'public'::regnamespace
  AND relname IN ('profiles', 'groupes', 'groupe_documents', 'groupe_acces');
-- Attendu : toutes à relrowsecurity = true (sauf groupes si pas de ENABLE RLS)
-- Note : groupes est protégé par REVOKE, pas par RLS directe

-- A3. Politiques sur groupe_documents

-- A8. 23 lignes dans groupe_acces
SELECT COUNT(*) FROM public.groupe_acces;
-- Attendu : 23

-- A9. Parents avec acces_cours = true (aucun perdu)
SELECT
  role,
  COUNT(*) FILTER (WHERE acces_cours = true)  AS avec_acces,
  COUNT(*) FILTER (WHERE acces_cours = false) AS sans_acces,
  COUNT(*) FILTER (WHERE acces_cours IS NULL) AS acces_null,
  COUNT(*)                                     AS total
FROM public.profiles
GROUP BY role ORDER BY role;
-- Attendu : parents avec_acces = nombre_initial, sans_acces = 0

-- A10. Statut du bucket groupe-docs
SELECT id, name, public AS est_public FROM storage.buckets WHERE id = 'groupe-docs';
-- Attendu avant Phase C : est_public = true (acceptable temporairement)
-- Attendu après Phase C : est_public = false

-- A11. ⭐ CRITIQUE v8 — Droits révoqués sur la TABLE groupes
SELECT grantee, privilege_type
FROM information_schema.role_table_grants
WHERE table_schema = 'public' AND table_name = 'groupes'
  AND grantee IN ('anon', 'authenticated')
ORDER BY grantee;
-- Attendu : 0 lignes (anon et authenticated révoqués)
-- ⚠️  Si des lignes apparaissent : Phase B incomplète, /rest/v1/groupes reste accessible

-- A12. ⭐ CRITIQUE v8 — Vue groupes_public présente et sans zoom_link
SELECT column_name FROM information_schema.columns
WHERE table_schema = 'public' AND table_name = 'groupes_public'
ORDER BY ordinal_position;
-- Attendu : id, categorie, jour, horaire, ordre  (PAS zoom_link, PAS devoirs)

-- A13. Droits GRANT sur la vue groupes_public
SELECT grantee, privilege_type
FROM information_schema.role_table_grants
WHERE table_schema = 'public' AND table_name = 'groupes_public'
  AND grantee IN ('anon', 'authenticated')
ORDER BY grantee;
-- Attendu : anon=SELECT, authenticated=SELECT

-- A14. Sauvegarde vérifiable : cohérence groupe_acces vs groupes
SELECT
  COUNT(*)                                                              AS total_groupes,
  COUNT(ga.groupe_id)                                                   AS groupes_dans_acces,
  COUNT(*) FILTER (WHERE g.zoom_link IS NOT NULL AND ga.zoom_link IS NULL) AS zoom_manquant,
  COUNT(*) FILTER (WHERE g.devoirs IS NOT NULL AND ga.devoirs IS NULL)     AS devoirs_manquant
FROM public.groupes g
LEFT JOIN public.groupe_acces ga ON ga.groupe_id = g.id;
-- Attendu : total_groupes = groupes_dans_acces, zoom_manquant = 0, devoirs_manquant = 0

-- A15. Documents généraux (table public.documents) — vérification manuelle
SELECT nom, storage_path, created_at FROM public.documents ORDER BY created_at DESC;
-- Vérifier visuellement qu'aucun document pédagogique n'est ici.
