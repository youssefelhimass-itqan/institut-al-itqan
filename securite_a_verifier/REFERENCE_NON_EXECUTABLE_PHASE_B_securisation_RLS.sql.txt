-- ════════════════════════════════════════════════════════════
-- BLOC 0 : Fonctions SECURITY DEFINER (créer ou remplacer)
-- Ces fonctions évitent la récursion RLS sur profiles.
-- ════════════════════════════════════════════════════════════

CREATE OR REPLACE FUNCTION public.get_my_role()
RETURNS text
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
  SELECT role FROM public.profiles WHERE id = auth.uid();
$$;

CREATE OR REPLACE FUNCTION public.get_my_acces_cours()
RETURNS boolean
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
  SELECT acces_cours FROM public.profiles WHERE id = auth.uid();
$$;

-- Vérification
SELECT routine_name, security_type
FROM information_schema.routines
WHERE routine_schema = 'public'
  AND routine_name IN ('get_my_role', 'get_my_acces_cours');
-- Attendu : 2 lignes, security_type = DEFINER


-- ════════════════════════════════════════════════════════════
-- BLOC 1 : Protection de groupes.zoom_link
--
-- PROBLÈME RÉEL (confirmé v7) :
--   • PostgreSQL n'accepte pas RLS sur une vue ordinaire.
--   • ALTER VIEW ... ENABLE ROW LEVEL SECURITY → erreur PostgreSQL.
--   • CREATE POLICY ... ON <vue> → erreur PostgreSQL.
--   • La politique "Lire groupes" USING(true) reste active sur la TABLE.
--   • PostgREST expose les tables ET les vues directement.
--   • Un appel /rest/v1/groupes?select=zoom_link contourne la vue.
--
-- SOLUTION v8 : REVOKE/GRANT au niveau SQL (couche inférieure à RLS)
--
-- MÉCANISME :
--   1. REVOKE SELECT ON public.groupes FROM anon, authenticated
--      → PostgREST ne peut plus servir la TABLE directement
--      → /rest/v1/groupes?select=zoom_link → 403 Forbidden
--   2. CREATE VIEW groupes_public (colonnes sûres uniquement)
--   3. GRANT SELECT ON public.groupes_public TO authenticated
--      → seule la vue est accessible via l'API
--      → /rest/v1/groupes_public?select=zoom_link → erreur "colonne inconnue"
--   4. La table reste accessible à postgres (service_role) pour les
--      opérations internes (AdminDashboard via service_role côté serveur)
--
-- NOTE : Le rôle "authenticator" de Supabase/PostgREST se connecte en tant
-- que "anon" ou "authenticated" selon le JWT. REVOKE sur ces deux rôles
-- ne suffit PAS si PUBLIC ou un rôle hérité possède encore SELECT.
-- ════════════════════════════════════════════════════════════

-- Étape 1a : Révoquer l'accès SELECT direct à la table groupes
-- (bloque /rest/v1/groupes pour anon et authenticated)
-- Révoquer aussi PUBLIC : les droits hérités de PUBLIC peuvent contourner
-- une révocation ciblée sur authenticated/anon.
REVOKE SELECT ON public.groupes FROM PUBLIC, anon, authenticated;

-- Échec explicite si l'un des deux rôles peut encore lire la table.
DO $$
BEGIN
  IF has_table_privilege('anon', 'public.groupes', 'SELECT')
     OR has_table_privilege('authenticated', 'public.groupes', 'SELECT')
  THEN
    RAISE EXCEPTION 'Accès SELECT effectif toujours présent sur public.groupes';
  END IF;
END $$;

-- Étape 1b : Supprimer la vue si elle existe déjà (idempotent)
DROP VIEW IF EXISTS public.groupes_public;

-- Étape 1c : Créer la vue avec uniquement les colonnes publiques
-- security_invoker = true : la vue s'exécute avec les droits de l'appelant
-- (si l'appelant n'a pas accès à groupes, la vue échoue aussi)
-- Mais ici la table groupes sera accessible uniquement via postgres/service_role,
-- donc la vue utilisera le definer (postgres) pour lire la table.
-- On utilise SECURITY DEFINER implicitement via la vue — voir note ci-dessous.
--
-- NOTE IMPORTANTE sur security_invoker vs security_definer pour les vues :
--   • security_invoker = true  → la vue utilise les droits de l'appelant
--     Problème : si authenticated n'a pas SELECT sur groupes (REVOKE ci-dessus),
--     la vue ne peut pas lire groupes non plus → erreur.
--   • Solution : laisser la vue en SECURITY DEFINER (défaut PostgreSQL pour les vues)
--     La vue s'exécute avec les droits du propriétaire (postgres/service_role)
--     qui, lui, peut lire groupes même après le REVOKE.
--     L'accès à la vue est contrôlé par GRANT/REVOKE sur la vue elle-même.
--
-- C'est l'architecture correcte : REVOKE sur la table + vue SECURITY DEFINER
-- + GRANT sur la vue aux rôles autorisés.
CREATE VIEW public.groupes_public AS
  SELECT
    id,
    categorie,
    jour,
    horaire,
    ordre
  FROM public.groupes;
-- ⚠️  Pas de WITH (security_invoker = true) — on veut SECURITY DEFINER (défaut)
--     pour que la vue puisse lire la table même après REVOKE sur authenticated.

-- Étape 1d : Accorder SELECT sur la vue aux rôles PostgREST
-- (la seule voie d'accès API pour les colonnes publiques)
GRANT SELECT ON public.groupes_public TO anon;
GRANT SELECT ON public.groupes_public TO authenticated;

-- Étape 1e : S'assurer que postgres (service_role) a toujours accès à la table
-- (normalement toujours le cas, mais explicite pour clarté)
GRANT ALL ON public.groupes TO postgres;

-- Vérifications
SELECT viewname, definition FROM pg_views
WHERE schemaname = 'public' AND viewname = 'groupes_public';
-- Attendu : vue présente, définition sans zoom_link ni devoirs

-- Les colonnes de la vue ne doivent pas contenir zoom_link
SELECT column_name FROM information_schema.columns
WHERE table_schema = 'public' AND table_name = 'groupes_public'
ORDER BY ordinal_position;
-- Attendu : id, categorie, jour, horaire, ordre  (PAS zoom_link, PAS devoirs)

-- Les privilèges sur la table groupes (anon et authenticated révoqués)
SELECT grantee, privilege_type
FROM information_schema.role_table_grants
WHERE table_schema = 'public' AND table_name = 'groupes'
ORDER BY grantee, privilege_type;
-- Attendu : postgres=ALL, anon=ABSENT, authenticated=ABSENT

-- Les privilèges sur la vue groupes_public
SELECT grantee, privilege_type
FROM information_schema.role_table_grants
WHERE table_schema = 'public' AND table_name = 'groupes_public'
ORDER BY grantee, privilege_type;
-- Attendu : anon=SELECT, authenticated=SELECT, postgres=ALL


-- ════════════════════════════════════════════════════════════
-- BLOC 2 : Profils — politique admin pour activer acces_cours
-- ════════════════════════════════════════════════════════════

ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "profiles_admin_select" ON public.profiles;
DROP POLICY IF EXISTS "profiles_admin_update"  ON public.profiles;

-- Admin : lire tous les profils
CREATE POLICY "profiles_admin_select"
ON public.profiles
FOR SELECT
TO authenticated
USING (public.get_my_role() = 'admin');

-- Admin : mettre à jour acces_cours et role
CREATE POLICY "profiles_admin_update"
ON public.profiles
FOR UPDATE
TO authenticated
USING  (public.get_my_role() = 'admin')
WITH CHECK (public.get_my_role() = 'admin');

-- Vérification
SELECT policyname FROM pg_policies
WHERE schemaname = 'public' AND tablename = 'profiles'
ORDER BY policyname;
-- Attendu : "Profil personnel" + "profiles_admin_select" + "profiles_admin_update"


-- ════════════════════════════════════════════════════════════
-- BLOC 3 : groupe_documents — restreindre à acces_cours = true
-- ════════════════════════════════════════════════════════════

ALTER TABLE public.groupe_documents ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Lire groupe_documents"            ON public.groupe_documents;
DROP POLICY IF EXISTS "groupe_documents_parent_select"   ON public.groupe_documents;
DROP POLICY IF EXISTS "groupe_documents_admin_all"        ON public.groupe_documents;

-- Parents autorisés : SELECT uniquement si acces_cours = true
CREATE POLICY "groupe_documents_parent_select"
ON public.groupe_documents
FOR SELECT
TO authenticated
USING (
  public.get_my_role()            = 'parent'
  AND public.get_my_acces_cours() IS TRUE
);

-- Admin : accès total
CREATE POLICY "groupe_documents_admin_all"
ON public.groupe_documents
FOR ALL
TO authenticated
USING  (public.get_my_role() = 'admin')
WITH CHECK (public.get_my_role() = 'admin');

-- Vérification : aucune politique USING(true) ne doit subsister
SELECT policyname, cmd, qual FROM pg_policies
WHERE schemaname = 'public' AND tablename = 'groupe_documents'
ORDER BY policyname;
-- Attendu : "groupe_documents_admin_all" + "groupe_documents_parent_select"
-- ⚠️  "Lire groupe_documents" NE DOIT PAS apparaître

SELECT policyname FROM pg_policies
WHERE schemaname = 'public' AND tablename = 'groupe_documents'
  AND qual = 'true';
-- Attendu : 0 lignes

ON storage.objects
FOR UPDATE
TO authenticated
USING  (bucket_id = 'groupe-docs' AND public.get_my_role() = 'admin')
WITH CHECK (bucket_id = 'groupe-docs' AND public.get_my_role() = 'admin');

-- BLOC 6 : Sauvegarde vérifiable avant Phase C
--
-- À exécuter AVANT PHASE_C_cleanup.sql pour s'assurer que
-- groupe_acces contient bien toutes les données de groupes.
-- ════════════════════════════════════════════════════════════

-- Comparaison groupe_acces vs groupes (vérifier la couverture)
SELECT
  g.id,
  g.categorie,
  g.horaire,
  g.jour,
  g.zoom_link            AS zoom_dans_groupes,
  ga.zoom_link           AS zoom_dans_groupe_acces,
  CASE
    WHEN g.zoom_link IS NOT DISTINCT FROM ga.zoom_link THEN 'OK'
    ELSE '⚠️ DIFFÉRENCE'
  END AS statut_zoom,
  g.devoirs              AS devoirs_dans_groupes,
  ga.devoirs             AS devoirs_dans_groupe_acces,
  CASE
    WHEN g.devoirs IS NOT DISTINCT FROM ga.devoirs THEN 'OK'
    ELSE '⚠️ DIFFÉRENCE'
  END AS statut_devoirs
FROM public.groupes g
LEFT JOIN public.groupe_acces ga ON ga.groupe_id = g.id
ORDER BY g.id;
-- Attendu : toutes les lignes avec statut_zoom = 'OK' et statut_devoirs = 'OK'
-- ⚠️  Si une différence apparaît : NE PAS exécuter Phase C. Corriger d'abord.

-- Résumé compact pour décision go/no-go
SELECT
  COUNT(*)                                                              AS total_groupes,
  COUNT(ga.groupe_id)                                                   AS groupes_dans_acces,
  COUNT(*) FILTER (WHERE g.zoom_link IS NOT DISTINCT FROM ga.zoom_link) AS zoom_coherent,
  COUNT(*) FILTER (WHERE g.zoom_link IS NOT NULL AND ga.zoom_link IS NULL) AS zoom_manquant_dans_acces
FROM public.groupes g
LEFT JOIN public.groupe_acces ga ON ga.groupe_id = g.id;
-- Attendu : total_groupes = groupes_dans_acces, zoom_manquant_dans_acces = 0
